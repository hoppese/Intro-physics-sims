\hwsolhead{Written HW 5}{Problem statements from the hand-written worksheet (not a
textbook exercise set): Ohm's law, the junction rule, the loop rule, and series \&
parallel resistors. Every battery is ideal ($r=0$). Problems 1--9 are Section 1
(required, 2 pts each); EC1--EC9 are Section 2 (extra credit, 1 pt each).}

%============================================================
\problem{1}{2}{A resistor has resistance $R=25\ \Omega$. A voltmeter placed across it
reads $\Delta V=15$ V. What current flows through the resistor?}

\model{Ohm's law relates the current, resistance, and voltage drop of a single
resistor.}

\solve{
\[
I = \frac{\Delta V}{R} = \frac{15\text{ V}}{25\ \Omega} = \boxed{0.60\text{ A}}.
\]}

%============================================================
\problem{2}{2}{Three wires meet at a junction: $2.0$ A and $3.5$ A flow in, and a
current $I$ flows out. Find $I$.}

\model{Kirchhoff's junction rule: charge doesn't pile up at a junction, so the total
current in equals the total current out.}

\solve{
\[
I = 2.0\text{ A}+3.5\text{ A} = \boxed{5.5\text{ A}}.
\]}

%============================================================
\problem{3}{2}{Four wires meet at a junction: $4.0$ A and $1.0$ A flow in, $2.0$ A and
$I$ flow out. Find $I$.}

\model{Junction rule again, now with currents on both sides of the balance.}

\solve{
\[
4.0\text{ A}+1.0\text{ A} = 2.0\text{ A}+I
\;\Rightarrow\;
I = \boxed{3.0\text{ A}}.
\]}

%============================================================
\problem{4}{2}{A single-loop circuit has a battery with $\varepsilon=20$ V and two
resistors in series. A voltmeter reads $\Delta V_1=12$ V across the first resistor. Find
$\Delta V_2$ across the second.}

\model{Kirchhoff's loop rule: going all the way around a loop, the total voltage gained
equals the total voltage lost. For two series resistors, their drops must add up to the
full emf.}

\solve{
\[
\varepsilon = \Delta V_1+\Delta V_2
\;\Rightarrow\;
\Delta V_2 = 20\text{ V}-12\text{ V} = \boxed{8.0\text{ V}}.
\]}

%============================================================
\problem{5}{2}{Same idea, different numbers: a single loop has $\varepsilon=24$ V and
two series resistors, with $\Delta V_1=9.0$ V. Find $\Delta V_2$.}

\model{Same loop rule as Problem 4.}

\solve{
\[
\Delta V_2 = \varepsilon-\Delta V_1 = 24\text{ V}-9.0\text{ V} = \boxed{15\text{ V}}.
\]}

%============================================================
\problem{6}{2}{Two resistors, $R_1=4.0\ \Omega$ and $R_2=8.0\ \Omega$, are in series
with a battery of $\varepsilon=18$ V. Find (a) the equivalent resistance, and (b) the
current supplied by the battery.}

\model{Series resistors add directly; the battery then drives that single equivalent
resistance.}

\solve{(a)
\[
R_{\text{eq}} = R_1+R_2 = 4.0\ \Omega+8.0\ \Omega = \boxed{12\ \Omega}.
\]
(b)
\[
I = \frac{\varepsilon}{R_{\text{eq}}} = \frac{18\text{ V}}{12\ \Omega} = \boxed{1.5\text{
A}}.
\]}

%============================================================
\problem{7}{2}{Three resistors, $R_1=2.0\ \Omega$, $R_2=3.0\ \Omega$, and
$R_3=5.0\ \Omega$, are in series with a battery of $\varepsilon=20$ V. Find (a) the
equivalent resistance, and (b) the current supplied by the battery.}

\model{Same idea as Problem 6, now with three series resistors.}

\solve{(a)
\[
R_{\text{eq}} = 2.0\ \Omega+3.0\ \Omega+5.0\ \Omega = \boxed{10\ \Omega}.
\]
(b)
\[
I = \frac{\varepsilon}{R_{\text{eq}}} = \frac{20\text{ V}}{10\ \Omega} = \boxed{2.0\text{
A}}.
\]}

%============================================================
\problem{8}{2}{Two resistors, $R_1=6.0\ \Omega$ and $R_2=3.0\ \Omega$, are in parallel
across a battery of $\varepsilon=12$ V. Find (a) the equivalent resistance, and (b) the
total current from the battery.}

\model{Parallel resistors combine by reciprocals; both branches see the full battery
voltage.}

\solve{(a)
\[
R_{\text{eq}} = \left(\frac{1}{6.0\ \Omega}+\frac{1}{3.0\ \Omega}\right)^{-1} =
\boxed{2.0\ \Omega}.
\]
(b)
\[
I_{\text{tot}} = \frac{\varepsilon}{R_{\text{eq}}} = \frac{12\text{ V}}{2.0\ \Omega} =
\boxed{6.0\text{ A}}.
\]}

\review{Check directly: $I_1=\varepsilon/R_1=12/6.0=2.0$ A and
$I_2=\varepsilon/R_2=12/3.0=4.0$ A, so $I_{\text{tot}}=I_1+I_2=6.0$ A --- matches.}

%============================================================
\problem{9}{2}{Two resistors are in parallel across a battery of $\varepsilon=12$ V.
$R_1=6.0\ \Omega$, and the \emph{total} current measured from the battery is $5.0$ A.
Find (a) the current through $R_2$, and (b) the value of $R_2$.}

\model{In parallel, each branch independently sees the full battery voltage, so
$R_1$'s current can be found on its own; the junction rule then gives whatever's left
for $R_2$.}

\solve{(a) $R_1$ alone carries
\[
I_1 = \frac{\varepsilon}{R_1} = \frac{12\text{ V}}{6.0\ \Omega} = 2.0\text{ A},
\]
so by the junction rule,
\[
I_2 = I_{\text{tot}}-I_1 = 5.0\text{ A}-2.0\text{ A} = \boxed{3.0\text{ A}}.
\]
(b)
\[
R_2 = \frac{\varepsilon}{I_2} = \frac{12\text{ V}}{3.0\text{ A}} = \boxed{4.0\ \Omega}.
\]}

\problem{EC1}{1}{A current of $0.40$ A flows through a resistor when a voltmeter across
it reads $8.0$ V. Find $R$.}

\solve{
\[
R = \frac{\Delta V}{I} = \frac{8.0\text{ V}}{0.40\text{ A}} = \boxed{20\ \Omega}.
\]}

%============================================================
\problem{EC2}{1}{At a junction, $6.0$ A flows in and splits into two outgoing wires;
one carries $2.5$ A. Find the other.}

\solve{
\[
I = 6.0\text{ A}-2.5\text{ A} = \boxed{3.5\text{ A}}.
\]}

%============================================================
\problem{EC3}{1}{Three wires carry current \emph{into} a junction: $1.5$ A, $2.0$ A, and
$2.5$ A. One wire carries it all back out. Find that current.}

\solve{
\[
I = 1.5\text{ A}+2.0\text{ A}+2.5\text{ A} = \boxed{6.0\text{ A}}.
\]}

%============================================================
\problem{EC4}{1}{A single loop has $\varepsilon=9.0$ V and \emph{three} series
resistors. Voltmeters across two of them read $2.0$ V and $3.0$ V. Find the reading
across the third.}

\solve{
\[
\Delta V_3 = \varepsilon-\Delta V_1-\Delta V_2 = 9.0\text{ V}-2.0\text{ V}-3.0\text{ V} =
\boxed{4.0\text{ V}}.
\]}

%============================================================
\problem{EC5}{1}{A single loop has $\varepsilon=30$ V and three series resistors.
$\Delta V_1=10$ V and $\Delta V_2=8.0$ V. Find $\Delta V_3$.}

\solve{
\[
\Delta V_3 = 30\text{ V}-10\text{ V}-8.0\text{ V} = \boxed{12\text{ V}}.
\]}

%============================================================
\problem{EC6}{1}{Two resistors, $R_1=3.0\ \Omega$ and $R_2=7.0\ \Omega$, are in series
with a battery of $\varepsilon=20$ V. Find the current supplied by the battery.}

\solve{
\[
I = \frac{\varepsilon}{R_1+R_2} = \frac{20\text{ V}}{10\ \Omega} = \boxed{2.0\text{ A}}.
\]}

%============================================================
\problem{EC7}{1}{Two resistors are in series with a battery of $\varepsilon=15$ V.
$R_1=2.0\ \Omega$, and the current measured in the circuit is $1.5$ A. Find $R_2$.}

\solve{The same current flows through both series resistors, so the equivalent
resistance is
\[
R_{\text{eq}} = \frac{\varepsilon}{I} = \frac{15\text{ V}}{1.5\text{ A}} = 10\ \Omega
\;\Rightarrow\;
R_2 = R_{\text{eq}}-R_1 = 10\ \Omega-2.0\ \Omega = \boxed{8.0\ \Omega}.
\]}

%============================================================
\problem{EC8}{1}{Two resistors, $R_1=8.0\ \Omega$ and $R_2=24\ \Omega$, are in parallel
across a battery of $\varepsilon=24$ V. Find the total current supplied by the battery.}

\solve{Each branch sees the full $24$ V independently:
\[
I_1 = \frac{24\text{ V}}{8.0\ \Omega} = 3.0\text{ A}, \qquad
I_2 = \frac{24\text{ V}}{24\ \Omega} = 1.0\text{ A}
\;\Rightarrow\;
I_{\text{tot}} = I_1+I_2 = \boxed{4.0\text{ A}}.
\]}

%============================================================
\problem{EC9}{1}{Two resistors, $R_1=5.0\ \Omega$ and $R_2=20\ \Omega$, are in parallel
across a battery. The \emph{total} current from the battery is measured to be $5.0$ A.
Find $\varepsilon$.}

\solve{Both branches share the same unknown voltage $\varepsilon$, so
\[
I_{\text{tot}} = \frac{\varepsilon}{R_1}+\frac{\varepsilon}{R_2} =
\varepsilon\left(\frac{1}{5.0\ \Omega}+\frac{1}{20\ \Omega}\right) =
\varepsilon\left(\frac{1}{4.0\ \Omega}\right)
\]
\[
\Rightarrow\;
\varepsilon = (5.0\text{ A})(4.0\ \Omega) = \boxed{20\text{ V}}.
\]}
