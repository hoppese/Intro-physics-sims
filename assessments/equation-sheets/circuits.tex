\section{Current, resistance \& Ohm's law \quad {\normalfont\small (Knight 26.1--26.5, 27.1--27.4)}}
\eqtab{
  \band{fundamental}
  \R{Current --- rate of charge flow}{I = \tfrac{\Delta Q}{\Delta t}}
  \R{Ohm's law\bul{for an ohmic resistor; also $\Delta V = IR$ across any $R$}}{I = \tfrac{\Delta V}{R}}
  \R{Resistor power\bul{rate electrical energy $\to$ heat/light}}{P = I\,\Delta V}
  \band{derived}
  \R{Resistor power, alternate forms}{P = I^{2}R = \dfrac{(\Delta V)^{2}}{R}}
}

%========================================================
\section{EMF \& real batteries \quad {\normalfont\small (Knight 28.2--28.3)}}
\eqtab{
  \band{fundamental}
  \R{EMF --- energy supplied per unit charge}{\mathcal{E} = \dfrac{W}{q} \quad (\text{units of volts})}
  \band{derived}
  \R{Terminal voltage of a real battery\bul{internal resistance $r$; equals $\mathcal{E}$ only when $I=0$}\bul{real batteries behave like ideal batteries with a single resistor in series}}{\Delta V_{\text{term}} = \mathcal{E} - I r}
  \R{Current from a single-loop source}{I = \dfrac{\mathcal{E}}{R + r}}
}

%========================================================
\section{Series \& parallel combinations \quad {\normalfont\small (Knight 28.4)}}
\eqtab{
  \band{fundamental}
  \R{What's shared in series}{\text{same } I \text{ through every element}}
  \R{What's shared in parallel}{\text{same } \Delta V \text{ across every element}}
  \R{Resistors in series\bul{voltages add; $R_{\text{s}}$ is bigger than any one}}{R_{\text{s}} = R_1 + R_2 + \cdots}
  \R{Resistors in parallel\bul{currents add; $R_{\text{p}}$ is smaller than any one}}{\dfrac{1}{R_{\text{p}}} = \dfrac{1}{R_1} + \dfrac{1}{R_2} + \cdots}
  \R{Capacitors\bul{combine the opposite way to resistors}}{C_{\text{p}} = C_1 + C_2 + \cdots,\qquad \dfrac{1}{C_{\text{s}}} = \textstyle\sum \dfrac{1}{C_i}}
  \band{derived}
  \R{Two resistors in series}{R_{\text{s}} = R_1 + R_2}
  \R{Two resistors in parallel}{R_{\text{p}} = \dfrac{R_1 R_2}{R_1 + R_2}}
  \R{Two capacitors in series}{C_{\text{s}} = \dfrac{C_1 C_2}{C_1 + C_2}}
  \R{Two capacitors in parallel}{C_{\text{p}} = C_1 + C_2}
}

%========================================================
\section{Kirchhoff's rules \quad {\normalfont\small (Knight 28.1, 28.5)}}
\eqtab{
  \band{fundamental}
  \R{Junction rule (charge conservation)}{\textstyle\sum I_{\text{in}} = \sum I_{\text{out}}}
  \R{Loop rule (energy conservation)\bul{$+$ across a source from $-$ to $+$; $-IR$ across a resistor in the direction of $I$}}{\textstyle\sum_{\text{loop}} \Delta V = 0}
}

\section{RC circuits \quad {\normalfont\small (Knight 28.6)}}
\eqtab{
  \band{fundamental}
  \R{Capacitor relation}{Q = C\,\Delta V_C}
  \R{Time constant}{\tau = RC}
  \band{derived}
  \R{Charging through a battery $\Delta V_B$\bul{starts at $Q=0$; $I_0 = \Delta V_B/R$}\bul{$Q,\Delta V_C$ climb to $63\%$ of final after one $\tau$; $I,\Delta V_R$ decay from their start}}{\begin{array}{@{}l@{}}
    Q(t) = C\,\Delta V_B\left(1 - e^{-t/\tau}\right),\quad I(t) = I_0\,e^{-t/\tau} \\[4pt]
    \Delta V_R(t) = \Delta V_B\,e^{-t/\tau},\quad \Delta V_C(t) = \Delta V_B\left(1 - e^{-t/\tau}\right)
  \end{array}}
  \addlinespace[7pt]
  \R{Discharging\bul{starts at $Q_0$, $\Delta V_0 = Q_0/C$, $I_0 = \Delta V_0/R$}\bul{all four fall to $37\%$ after one $\tau$}}{\begin{array}{@{}l@{}}
    Q(t) = Q_0\,e^{-t/\tau},\quad I(t) = I_0\,e^{-t/\tau} \\[4pt]
    \Delta V_R(t) = \Delta V_0\,e^{-t/\tau},\quad \Delta V_C(t) = \Delta V_0\,e^{-t/\tau}
  \end{array}}
}
