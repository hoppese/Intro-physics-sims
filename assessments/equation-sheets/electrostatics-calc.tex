\section{Doing the calculus: continuous distributions \& Gauss's law \quad {\normalfont\small (Knight 23.3--23.6, 24.1--24.3, 25.4)}}
\eqtab{
  \band{fundamental}
  \R{Potential of a continuous distribution --- chop, add up\newline\footnotesize($V$ is a scalar: no components, no cancellation to track)}{V = \displaystyle\int dV,\qquad dV = \dfrac{1}{4\pi\varepsilon_0}\dfrac{dq}{r}}
  \R{Evaluating a Gaussian-surface integral\newline\footnotesize(pick a surface where $E$ is constant and either $\parallel$ or $\perp$ to $d\vec A$ everywhere)}{\oint \vec E\cdot d\vec A = E\!\oint dA = EA \ \Rightarrow\ E=\dfrac{Q_{\text{enc}}}{\varepsilon_0 A}}
  \band{derived}
  \R{Ring of charge $Q$, radius $R$, on axis at $z$}{V = \dfrac{1}{4\pi\varepsilon_0}\dfrac{Q}{\sqrt{z^2+R^2}}}
  \R{Disk of charge $Q$, radius $R$, on axis at $z$\newline\footnotesize($\eta=Q/\pi R^2$)}{V = \dfrac{\eta}{2\varepsilon_0}\Big(\sqrt{z^2+R^2}-z\Big)}
}

\section{Bonus: how $\vec F$, $U$, $\vec E$, $V$ connect \quad {\normalfont\small might appear as an extra-credit problem only}}
\eqtab{
  \bonusband{Force $\leftrightarrow$ energy}
  \R{Force is the (negative) slope of $U$}{F_x = -\dfrac{dU}{dx}\,,\qquad \vec F = -\vec\nabla U}
  \R{$U$ is (minus) the work integral of $\vec F$\newline\footnotesize(same statement as $W_{\text{elec}}=-\Delta U$ above, written as a line integral)}{U_f - U_i = -\displaystyle\int_i^f \vec F\cdot d\vec s}
  \bonusband{Field $\leftrightarrow$ potential}
  \R{Field is the (negative) slope of $V$}{E_x = -\dfrac{dV}{dx}\,,\qquad \vec E = -\vec\nabla V}
  \R{$V$ is (minus) the line integral of $\vec E$}{V_f - V_i = -\displaystyle\int_i^f \vec E\cdot d\vec s}
}
{\footnotesize\color{bonus} The two rows are the same relationship twice: $U$ and $V$ are what you integrate a force/field \emph{against}
to get; $\vec F$ and $\vec E$ are what falls out when you differentiate $U$ or $V$. Swap $U\to V$ and $\vec F \to \vec E$
(divide through by $q$) and the left column becomes the right column.\newline
$\vec\nabla \equiv \dfrac{\partial}{\partial x}\hat x + \dfrac{\partial}{\partial y}\hat y + \dfrac{\partial}{\partial z}\hat z$
--- the gradient operator, in Cartesian coordinates.}
