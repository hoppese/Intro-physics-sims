%========================================================
\section{SHM --- kinematics \quad {\normalfont\small (Knight 15.1--15.2)}}
\eqtab{
  \band{fundamental}
  \R{Position (the definition of SHM)}{x(t) = A\cos(\omega t + \varphi_0)}
  \R{Angular frequency \& frequency}{\omega = 2\pi f,\qquad f = 1/T}
  \band{derived}
  \R{Velocity}{v_x(t) = -\,\omega A\sin(\omega t + \varphi_0),\qquad v_{\max} = \omega A}
  \R{Acceleration}{a_x(t) = -\,\omega^{2} A\cos(\omega t + \varphi_0) = -\,\omega^{2}x}
  \R{Period from $\omega$}{T = 2\pi/\omega}
}

%========================================================
\section{SHM --- dynamics \& energy \quad {\normalfont\small (Knight 15.3--15.4)}}
\eqtab{
  \band{fundamental}
  \R{Restoring force (Hooke's law)}{F_x = -\,kx}
  \R{Angular frequency, mass on a spring}{\omega = \sqrt{k/m}}
  \R{Total mechanical energy (constant)}{E = \tfrac{1}{2}kA^{2}}
  \band{derived}
  \R{Period, mass on a spring}{T = 2\pi\sqrt{m/k}}
  \R{Energy at any instant}{\tfrac{1}{2}mv^{2} + \tfrac{1}{2}kx^{2} = \tfrac{1}{2}kA^{2} = \tfrac{1}{2}m\,v_{\max}^{2}}
  \R{Speed as a function of position}{v(x) = \pm\,\omega\sqrt{A^{2}-x^{2}}}
}

%========================================================
\section{Vertical oscillations \& the pendulum \quad {\normalfont\small (Knight 15.5--15.6)}}
\eqtab{
  \band{fundamental}
  \R{Simple pendulum (small angle)}{\omega = \sqrt{g/L}}
  \R{Hanging spring}{\text{still SHM, same } \omega=\sqrt{k/m}\,;\ \text{equilibrium shifts by } mg/k}
  \band{derived}
  \R{Pendulum period}{T = 2\pi\sqrt{L/g}}
}

%========================================================
\section{Traveling waves \quad {\normalfont\small (Knight 16.1--16.3)}}
\eqtab{
  \band{fundamental}
  \R{Wave (sinusoidal)}{y(x,t) = A\sin(kx - \omega t + \varphi_0)}
  \R{}{\text{$kx-\omega t$: moves $+x$ \qquad $kx+\omega t$: moves $-x$}}
  \R{Wave number}{k = 2\pi/\lambda}
  \R{Wave speed}{v = \lambda f}
  \band{derived}
  \R{Wave speed --- equivalent forms}{v = \lambda/T = \omega/k}
  \R{Transverse speed of a string element}{v_{y,\max} = \omega A \quad(\text{\emph{not} the wave speed } v)}
}

%========================================================
\section{Waves on a string \& in pipes \quad {\normalfont\small (Knight 17.1--17.2)}}
\eqtab{
  \band{fundamental}
  \R{Wave speed on a string}{v = \sqrt{T_s/\mu}\,,\qquad \mu = m/L}
  \R{Standing wavelengths --- ends alike\newline\footnotesize(both fixed, or pipe open--open / closed--closed)}{\lambda_n = 2L/n\,,\qquad n = 1,2,3,\dots}
  \R{Standing wavelengths --- one node end,\newline one antinode end \footnotesize(string fixed--free, pipe open--closed)}{\lambda_n = 4L/n\,,\qquad n = 1,3,5,\dots}
  \band{derived}
  \R{Harmonic frequencies}{f_n = \dfrac{v}{\lambda_n} = \dfrac{n v}{2L} = n f_1 \quad\left(\text{or } \dfrac{n v}{4L},\ n \text{ odd}\right)}
  \R{Node-to-node (or antinode) spacing}{\Delta x = \lambda/2}
}

%========================================================
\section{Superposition \& interference \quad {\normalfont\small (Knight 16.3, 17.3--17.5)}}
\eqtab{
  \R{Principle of superposition}{y(x,t) = y_1(x,t) + y_2(x,t)}
  \R{Two in-phase sources --- constructive}{\Delta r = m\lambda,\qquad m = 0,1,2,\dots}
  \R{Two in-phase sources --- destructive}{\Delta r = \left(m + \tfrac{1}{2}\right)\lambda}
}
