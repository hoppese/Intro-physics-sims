\section{Doing the calculus: Biot--Savart \& Amp\`ere's law \quad {\normalfont\small (Knight 29.4--29.6)}}
\eqtab{
  \band{fundamental}
  \R{Biot--Savart law --- field of a current element\newline\footnotesize(the calculus limit of Sec.~2's chop-and-add rule: $\Delta\vec L\to d\vec l$, sum $\to$ integral)}{d\vec B = \dfrac{\mu_0}{4\pi}\,\dfrac{I\,d\vec l \times \hat r}{r^{2}}\,,\qquad \vec B=\displaystyle\int d\vec B}
  \R{Amp\`ere's law\newline\footnotesize(any closed loop; $I_{\text{enc}}$ is the \emph{signed} current through the loop -- this is the formal version of Sec.~3's "only enclosed current matters")}{\oint \vec B \cdot d\vec l = \mu_0 I_{\text{enc}}}
  \band{derived}
  \R{Evaluating an Amperian-loop integral\newline\footnotesize(pick a loop where $B$ is constant and $\parallel$ to $d\vec l$ everywhere)}{\oint \vec B\cdot d\vec l = B\!\oint dl = B\,L_{\text{loop}} \ \Rightarrow\ B=\dfrac{\mu_0 I_{\text{enc}}}{L_{\text{loop}}}}
  \R{Current entirely outside the loop}{\oint \vec B \cdot d\vec l = 0}
}

\section{Bonus: torque and orientation energy \quad {\normalfont\small might appear as an extra-credit problem only}}
\eqtab{
  \bonusband{torque $\leftrightarrow$ energy}
  \R{Torque is the (negative) slope of $U$ with respect to angle\newline\footnotesize(same relationship as $F_x=-dU/dx$ on the electrostatics sheet, just swap position for angle)}{\tau_\theta = -\dfrac{dU}{d\theta}}
}
{\footnotesize\color{bonus} Check it: $U=-\mu B\cos\theta \Rightarrow -dU/d\theta = -\mu B\sin\theta$, which matches $|\vec\tau|=\mu B\sin\theta$ (Sec.~4) up to the sign convention for which way $\theta$ increases.}
